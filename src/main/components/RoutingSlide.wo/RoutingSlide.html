Routing

Application.java

public Application() {
	ERXRouter.declare( routes -> {
		routes.map( "/", Main.class );
		routes.map( "/books/{book}", BookPage.class );
		routes.map( "/feed", ri -> feed( ri ) );
	} );
}

BookPage.java

public class BookPage extends ERXComponent {

	public Integer book; // set from the URL
}

Routes.java

public interface Routes {

	default Object routes() {
		return RouteKeys.of( Routes.class );
	}

	PlainRoute book = Route.plain();
}

A link, in any template

<wo:route to="$routes.book" :book="$book.id">
	<wo:str value="$book.title" />
</wo:route>
